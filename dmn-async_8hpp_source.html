119 template <class Rep, class Period>
120 static auto toClockDuration(std::chrono::duration<Rep, Period> duration)
121 -> Clock::duration {
122 using ClockRep = Clock::duration::rep;
123 static_assert(
124 std::is_integral_v<ClockRep> &&
125 std::numeric_limits<ClockRep>::is_signed,
126 "steady_clock duration representation must be signed integer");
127 static_assert(std::numeric_limits<ClockRep>::radix == 2,
128 "steady_clock duration representation must be binary");
129
130 const std::chrono::duration<long double, Clock::period> clock_ticks{
131 duration};
132 const long double rounded_ticks = std::ceil(clock_ticks.count());
133 const auto max_exponent = std::numeric_limits<long double>::max_exponent;
134 const long double exclusive_max =
135 std::numeric_limits<ClockRep>::digits < max_exponent
136 ? std::ldexp(1.0L, std::numeric_limits<ClockRep>::digits)
137 : std::numeric_limits<long double>::infinity();
138 if (!std::isfinite(rounded_ticks) || rounded_ticks < 0 ||
139 rounded_ticks >= exclusive_max) {
140 throw std::overflow_error("Dmn_Async delay exceeds steady_clock range");
141 }
142
143 return Clock::duration{static_cast<ClockRep>(rounded_ticks)};
144 }
145
146 static auto nextDeadline(Clock::duration delay) -> Clock::time_point {
147 using Rep = Clock::duration::rep;
148 const auto now = Clock::now();
149 const auto now_ticks = now.time_since_epoch().count();
150 const auto delay_ticks = delay.count();
151
152 if (now_ticks > 0 &&
153 delay_ticks > std::numeric_limits<Rep>::max() - now_ticks) {
154 throw std::overflow_error(
155 "Dmn_Async deadline exceeds steady_clock range");
156 }
157
158 return Clock::time_point{Clock::duration{now_ticks + delay_ticks}};
159 }

364Dmn_Async<QueueType>::Dmn_Async(std::string_view name) : m_name{name} {
365 m_pipe = std::make_unique<BasePipe>(
366 m_name,
367 [](std::shared_ptr<Dmn_Async::Dmn_Async_Handle> task) -> void {
368 try {
369 if (task->m_fnc) {
370 task->m_fnc();

372
373 task->m_p.set_value();
374 } catch (...) {
375 task->m_p.set_exception(std::current_exception());

401template <class Rep, class Period, typename Callable, typename... Args>
402auto Dmn_Async<QueueType>::addExecTaskAfterWithWait(
403 const std::chrono::duration<Rep, Period> &duration, Callable &&func,
404 Args &&...args) -> std::shared_ptr<Dmn_Async_Handle> {
405 if (duration < std::chrono::duration<Rep, Period>::zero()) {
406 throw std::invalid_argument("Dmn_Async delay must not be negative");
407 }
408
409 const auto clock_duration = toClockDuration(duration);
410 const auto deadline = nextDeadline(clock_duration);
411
412 auto bound_task = [f = std::forward<Callable>(func),
413 ... captured_args = std::forward<Args>(args)]() mutable {
414 std::invoke(std::move(f), std::move(captured_args)...);
415 };
416
417 auto task_shared_ptr =
418 std::make_shared<Dmn_Async::Dmn_Async_Handle>(std::move(bound_task));
419 auto task_ret = task_shared_ptr;
420
421 this->m_pipe->writeAt(deadline, task_shared_ptr);
422
423 return task_ret;
424}

453auto Dmn_Async<QueueType>::addExecTaskFuture(Callable &&func, Args &&...args)
454 -> std::future<
455 std::invoke_result_t<std::decay_t<Callable>, std::decay_t<Args>...>> {
456 using Result =
457 std::invoke_result_t<std::decay_t<Callable>, std::decay_t<Args>...>;
458 static_assert(!std::is_reference_v<Result>,
459 "Dmn_Async future tasks must return an owned value or void");
460
461 auto promise = std::make_shared<std::promise<Result>>();
462 auto future = promise->get_future();
463 auto completionTask = [promise, callable = std::forward<Callable>(func),
464 ... captured_args =
465 std::forward<Args>(args)]() mutable {
466 try {
467 if constexpr (std::is_void_v<Result>) {
468 std::invoke(std::move(callable), std::move(captured_args)...);
469 promise->set_value();
470 } else {
471 promise->set_value(
472 std::invoke(std::move(callable), std::move(captured_args)...));
473 }
474 } catch (...) {
475 promise->set_exception(std::current_exception());
476 }
477 };
478
479 this->addExecTaskWithWait(std::move(completionTask));
480
481 return future;
