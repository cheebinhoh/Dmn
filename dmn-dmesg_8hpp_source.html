734inline auto Dmn_DMesg::finalizeHandlerRegistration(
735 std::shared_ptr<Dmn_DMesg::Dmn_DMesgHandler> handler) -> HandlerType {
736 auto handlerProxy = Dmn_DMesg::Dmn_DMesgHandlerProxy();
737
738 handler->m_owner = this;
739 this->registerSubscriber(handler);
740 handlerProxy.m_handler = handler;
741
742 // Execute registration and playback on the publisher's async thread
743 auto waitHandler = this->addExecTaskWithWait([this, handler]() {
744 this->m_handlers.push_back(handler);
745 this->playbackLastTopicDMesgPbInternal();
746 handler->setAfterInitialPlayback();
747 });
748
749 waitHandler->wait();
750
751 return handlerProxy;
752}

754template <class... U> auto Dmn_DMesg::openHandler(U &&...arg) -> HandlerType {
755 // This function:
756 // - constructs a handler
757 // - registers the handler as a subscriber
758 // - wires handler<->subscriber<->publisher links
759 // - schedules an async task on the publisher's singleton async thread to:
760 // * add the handler to the internal list
761 // * playback last-known messages per topic
762 // * mark handler as initialized after playback
763 //
764 // The use of the publisher's singleton async context keeps most operations
765 // mutex-free for the hot paths (publish/notify).
766
767 std::shared_ptr<Dmn_DMesg::Dmn_DMesgHandler> handler =
768 std::make_shared<Dmn_DMesg::Dmn_DMesgHandler>(std::forward<U>(arg)...);
769
770 return finalizeHandlerRegistration(handler);
771}
