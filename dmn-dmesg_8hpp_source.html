865inline auto Dmn_DMesg::finalizeHandlerRegistration(
866 std::shared_ptr<Dmn_DMesg::Dmn_DMesgHandler> handler) -> HandlerType {
867 auto handlerProxy = Dmn_DMesg::Dmn_DMesgHandlerProxy();
868
869 handler->m_owner = this;
870 this->registerSubscriber(handler);
871 handlerProxy.m_handler = handler;
872
873 // Queue the handler's playback-complete marker after playback deliveries.
874 auto waitHandler = this->addExecTaskWithWait([this, handler]() {
875 this->m_handlers.push_back(handler);
876 this->m_handler_states.try_emplace(handler.get());
877 this->playbackLastTopicDMesgPbInternal();
878 handler->setAfterInitialPlayback();
879 });
880
881 waitHandler->wait();
882
883 return handlerProxy;
884}

886template <class... U>
887 requires(sizeof...(U) != 1 ||
888 (!std::is_same_v<std::remove_cvref_t<U>, Dmn_DMesg::HandlerSpec> &&
889 ...))
890auto Dmn_DMesg::openHandler(U &&...arg) -> HandlerType {
891 // This function:
892 // - constructs a handler
893 // - registers the handler as a subscriber
894 // - wires handler<->subscriber<->publisher links
895 // - schedules an async task on the publisher's singleton async thread to:
896 // * add the handler and initialize its publisher-owned state snapshot
897 // * playback last-known messages per topic
898 // * queue the handler initialization marker after playback deliveries
899 //
900 // Publisher-owned handler snapshots and registration/playback are
901 // serialized on the publisher context. Each handler receives snapshots on
902 // its own async context for public state access and callbacks.
903
904 std::shared_ptr<Dmn_DMesg::Dmn_DMesgHandler> handler =
905 std::make_shared<Dmn_DMesg::Dmn_DMesgHandler>(std::forward<U>(arg)...);
906
907 return finalizeHandlerRegistration(handler);
908}
