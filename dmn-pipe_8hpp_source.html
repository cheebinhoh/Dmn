396template <typename T, typename QueueType>
397Dmn_Pipe<T, QueueType>::Dmn_Pipe(std::string_view name, Dmn_Pipe::Task fn,
398 size_t count, long timeout,
399 bool enable_scheduled_writes)
400 : Dmn_Proc{name}, m_hasFn{nullptr != fn},
401 m_scheduled_writes_enabled{enable_scheduled_writes} {
402 if (m_scheduled_writes_enabled && !fn) {
403 throw std::invalid_argument(
404 "Dmn_Pipe scheduled writes require a processing task");
405 }
406
407 if (fn) {
408 if (m_scheduled_writes_enabled) {
409 if (!exec([this, fn = std::move(fn)]() {

429 } catch (...) {

431 }

469template <typename T, typename QueueType>
470auto Dmn_Pipe<T, QueueType>::readAndProcess(Dmn_Pipe::Task fn, size_t count,
471 long timeout) -> size_t {
472 auto dataList = this->pop(count, timeout);
473
474 size_t processedCount = dataList.size();
475
476 if (fn) {
477 for (auto &item : dataList) {
478 Dmn_Proc::testcancel();
479
480 fn(std::move_if_noexcept(item));
481 }
482 }
483
484 std::unique_lock<std::mutex> lock(m_mutex);
485
486 m_count += processedCount;
487
488 m_empty_cond.notify_all();
489
490 return processedCount;
491}

595template <typename T, typename QueueType>
596void Dmn_Pipe<T, QueueType>::scheduledProcessingLoop(const Dmn_Pipe::Task &fn) {
597 while (true) {
598 std::optional<T> item{};
599
600 {
601 std::unique_lock<std::mutex> lock(m_mutex);
602
603 while (true) {
604 auto immediate_item = QueueType::popNoWait();
605 if (immediate_item) {
606 item.emplace(std::move_if_noexcept(*immediate_item));
607
608 break;
609 }
610
611 if (!m_scheduled_queue.empty()) {
612 const auto deadline = m_scheduled_queue.begin()->first;
613 const auto now = std::chrono::steady_clock::now();
614
615 if (deadline <= now) {
616 item.emplace(std::move(m_scheduled_queue.begin()->second));
617 m_scheduled_queue.erase(m_scheduled_queue.begin());
618
619 break;
620 }
621
622 m_work_cond.wait_until(lock, deadline);
623
624 continue;
625 }
626
627 if (isShutdown()) {
628 return;
629 }
630
631 m_work_cond.wait(lock);
632 }
633 }
634
635 fn(std::move_if_noexcept(*item));
636
637 {
638 std::lock_guard<std::mutex> lock(m_mutex);
639 ++m_count;
640 }
641
642 m_empty_cond.notify_all();
643 }
