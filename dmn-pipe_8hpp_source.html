433template <typename T, typename QueueType>
434Dmn_Pipe<T, QueueType>::Dmn_Pipe(std::string_view name, Dmn_Pipe::Task fn,
435 size_t count, long timeout,
436 bool enable_scheduled_writes)
437 : Dmn_Proc{name}, m_hasFn{nullptr != fn},
438 m_scheduled_writes_enabled{enable_scheduled_writes} {
439 if (m_scheduled_writes_enabled && !fn) {
440 throw std::invalid_argument(
441 "Dmn_Pipe scheduled writes require a processing task");
442 }
443
444 if (fn) {
445 bool execSuccess{};
446
447 if (m_scheduled_writes_enabled) {
448 execSuccess = exec([this, fn = std::move(fn)]() {
449 try {
450 scheduledProcessingLoop(fn);
451 }
452#if defined(__GLIBCXX__)
453 catch (const __cxxabiv1::__forced_unwind &) {
454 throw;
455 }
456#endif
457 catch (...) {
458 recordProcessingFailure(std::current_exception());
459 }
460 });
461 } else {
462 execSuccess = exec([this, fn = std::move(fn), count, timeout]() {
463 while (true) {
464 Dmn_Proc::yield();
465
466 try {
467 readAndProcess(fn, count, timeout);
468 }
469#if defined(__GLIBCXX__)
470 catch (const __cxxabiv1::__forced_unwind &) {
471 throw;
472 }
473#endif
474 catch (...) {
475 recordProcessingFailure(std::current_exception());
476
477 break;
478 }
479
480 // This check runs inside the worker after startup and determines
481 // when its processing loop should exit.
482 if (isShutdown() || hasProcessingFailure()) {
483 break;
484 }
485 }
486 });
487 }
488
489 if (!execSuccess) {
490 // Startup failure leaves no worker to process queued work, so fail
491 // construction instead of returning an unusable pipe.
492 throw std::runtime_error("failed to start Dmn_Pipe worker");
493 }
494 }

525template <typename T, typename QueueType>
526auto Dmn_Pipe<T, QueueType>::readAndProcess(Dmn_Pipe::Task fn, size_t count,
527 long timeout) -> size_t {
528 auto dataList{this->pop(count, timeout)};
529
530 size_t processedCount{};
531
532 try {
533 if (fn) {
534 for (auto &item : dataList) {
535 Dmn_Proc::testcancel();
536
537 fn(std::move_if_noexcept(item));
538 ++processedCount;
539 }
540 } else {
541 processedCount = dataList.size();
542 }
543 }
544#if defined(__GLIBCXX__)
545 catch (const __cxxabiv1::__forced_unwind &) {
546 throw;
547 }
548#endif
549 catch (...) {
550 const auto failure{std::current_exception()};
551
552 {
553 std::lock_guard<std::mutex> lock(m_mutex);
554 m_count += processedCount;
555
556 if (!m_processing_exception) {
557 m_processing_exception = failure;
558 }
559 }
560
561 m_empty_cond.notify_all();
562 m_work_cond.notify_all();
563
564 throw;
565 }
566
567 {
568 std::lock_guard<std::mutex> lock(m_mutex);
569 m_count += processedCount;
570 }
571
572 m_empty_cond.notify_all();
573

687template <typename T, typename QueueType>
688void Dmn_Pipe<T, QueueType>::scheduledProcessingLoop(const Dmn_Pipe::Task &fn) {
689 while (true) {
690 std::optional<T> item{};
691
692 {
693 std::unique_lock<std::mutex> lock(m_mutex);
694
695 while (true) {
696 if (m_processing_exception) {
697 return;
698 }
699
700 auto immediate_item{QueueType::popNoWait()};
701 if (immediate_item) {
702 item.emplace(std::move_if_noexcept(*immediate_item));
703
704 break;
705 }
706
707 if (!m_scheduled_queue.empty()) {
708 const auto deadline{m_scheduled_queue.begin()->first};
709 const auto now{std::chrono::steady_clock::now()};
710
711 if (deadline <= now) {
712 item.emplace(std::move(m_scheduled_queue.begin()->second));
713 m_scheduled_queue.erase(m_scheduled_queue.begin());
714
715 break;
716 }
717
718 m_work_cond.wait_until(lock, deadline);
719
720 continue;
721 }
722
723 if (isShutdown()) {
724 return;
725 }
726
727 m_work_cond.wait(lock);
728 }
729 }
730
731 fn(std::move_if_noexcept(*item));
732
733 {
734 std::lock_guard<std::mutex> lock(m_mutex);
735 ++m_count;
736 }
737
738 m_empty_cond.notify_all();
739 }
