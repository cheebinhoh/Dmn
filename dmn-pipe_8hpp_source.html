428template <typename T, typename QueueType>
429Dmn_Pipe<T, QueueType>::Dmn_Pipe(std::string_view name, Dmn_Pipe::Task fn,
430 size_t count, long timeout,
431 bool enable_scheduled_writes)
432 : Dmn_Proc{name}, m_hasFn{nullptr != fn},
433 m_scheduled_writes_enabled{enable_scheduled_writes} {
434 if (m_scheduled_writes_enabled && !fn) {
435 throw std::invalid_argument(
436 "Dmn_Pipe scheduled writes require a processing task");
437 }
438
439 if (fn) {
440 bool execSuccess{};
441
442 if (m_scheduled_writes_enabled) {
443 execSuccess = exec([this, fn = std::move(fn)]() {
444 try {
445 scheduledProcessingLoop(fn);
446 } catch (...) {
447 recordProcessingFailure(std::current_exception());
448 }
449 });
450 } else {
451 execSuccess = exec([this, fn = std::move(fn), count, timeout]() {
452 while (true) {
453 Dmn_Proc::yield();
454
455 try {
456 readAndProcess(fn, count, timeout);
457 } catch (...) {
458 recordProcessingFailure(std::current_exception());
459
460 break;
461 }
462
463 // This check runs inside the worker after startup and determines
464 // when its processing loop should exit.
465 if (isShutdown() || hasProcessingFailure()) {
466 break;
467 }
468 }
469 });
470 }
471
472 if (!execSuccess) {
473 // Startup failure leaves no worker to process queued work, so fail
474 // construction instead of returning an unusable pipe.
475 throw std::runtime_error("failed to start Dmn_Pipe worker");
476 }
477 }

664template <typename T, typename QueueType>
665void Dmn_Pipe<T, QueueType>::scheduledProcessingLoop(const Dmn_Pipe::Task &fn) {
666 while (true) {
667 std::optional<T> item{};
668
669 {
670 std::unique_lock<std::mutex> lock(m_mutex);
671
672 while (true) {
673 if (m_processing_exception) {
674 return;
675 }
676
677 auto immediate_item = QueueType::popNoWait();
678 if (immediate_item) {
679 item.emplace(std::move_if_noexcept(*immediate_item));
680
681 break;
682 }
683
684 if (!m_scheduled_queue.empty()) {
685 const auto deadline = m_scheduled_queue.begin()->first;
686 const auto now = std::chrono::steady_clock::now();
687
688 if (deadline <= now) {
689 item.emplace(std::move(m_scheduled_queue.begin()->second));
690 m_scheduled_queue.erase(m_scheduled_queue.begin());
691
692 break;
693 }
694
695 m_work_cond.wait_until(lock, deadline);
696
697 continue;
698 }
699
700 if (isShutdown()) {
701 return;
702 }
703
704 m_work_cond.wait(lock);
705 }
706 }
707
708 fn(std::move_if_noexcept(*item));
709
710 {
711 std::lock_guard<std::mutex> lock(m_mutex);
712 ++m_count;
713 }
714
715 m_empty_cond.notify_all();
716 }
