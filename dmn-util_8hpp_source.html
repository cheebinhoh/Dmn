65inline std::vector<UChar> stringToUnicode(const std::string_view value) {
66 if (value.size() > static_cast<std::size_t>(INT32_MAX)) {
67 throw std::length_error{"stringCompare input exceeds ICU's size limit"};
68 }
69
70 const auto *utf8 = value.empty() ? "" : value.data();
71 const auto length = static_cast<int32_t>(value.size());
72 UErrorCode status = U_ZERO_ERROR;
73 int32_t unicodeLength = 0;
74
75 u_strFromUTF8(nullptr, 0, &unicodeLength, utf8, length, &status);
76 if (status != U_BUFFER_OVERFLOW_ERROR && U_FAILURE(status)) {
77 throw std::invalid_argument{"stringCompare input is not valid UTF-8"};
78 }
79
80 status = U_ZERO_ERROR;
81 std::vector<UChar> result(static_cast<std::size_t>(unicodeLength) + 1);
82
83 u_strFromUTF8(result.data(), static_cast<int32_t>(result.size()),
84 &unicodeLength, utf8, length, &status);
85 if (U_FAILURE(status)) {
86 throw std::invalid_argument{"stringCompare input is not valid UTF-8"};
87 }
88
89 result.resize(static_cast<std::size_t>(unicodeLength));
90
91 return result;
92}

102inline std::vector<UChar> normalizeNfd(const std::vector<UChar> &value,
103 const UNormalizer2 *normalizer) {
104 const auto *source = value.empty() ? u"" : value.data();
105 const auto length = static_cast<int32_t>(value.size());
106 UErrorCode status = U_ZERO_ERROR;
107
108 const auto resultLength =
109 unorm2_normalize(normalizer, source, length, nullptr, 0, &status);
110 if (status != U_BUFFER_OVERFLOW_ERROR && U_FAILURE(status)) {
111 throw std::runtime_error{"ICU failed to normalize a string"};
112 }

114 status = U_ZERO_ERROR;

122 }

171 UErrorCode status = U_ZERO_ERROR;
172 const auto *normalizer = unorm2_getNFDInstance(&status);
173 if (U_FAILURE(status)) {
174 throw std::runtime_error{"ICU failed to initialize Unicode normalization"};
175 }
176
