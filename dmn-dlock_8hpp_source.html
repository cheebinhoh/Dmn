252 auto canonicalize() -> void {
253 std::sort(m_entries.begin(), m_entries.end(),
254 [](const Dmn_DLock_Entry &lhs, const Dmn_DLock_Entry &rhs) {
255 const auto lhs_key = lhs.canonicalKey();
256 const auto rhs_key = rhs.canonicalKey();
257 return lhs_key < rhs_key;
258 });

261 auto validate() const -> bool {
262 if (m_domain.empty()) {
263 return false;
264 }
265
266 for (const auto &entry : m_entries) {
267 if (!entry.isValid()) {
268 return false;
269 }
270 if (!entry.m_range.isValid()) {
271 return false;
272 }
273 }
274
275 for (std::size_t i = 0; i < m_entries.size(); ++i) {
276 for (std::size_t j = i + 1; j < m_entries.size(); ++j) {
277 const auto &lhs = m_entries[i];
278 const auto &rhs = m_entries[j];
279 const bool same_domain = lhs.m_domain == rhs.m_domain;
280 const bool granted_overlap =
281 lhs.m_granted && rhs.m_granted && lhs.m_range.overlaps(rhs.m_range);
282 if (same_domain && granted_overlap) {
283 return false;
284 }
285 }
286 }
287
288 return true;

291 auto classifyGrantCandidate(const Dmn_DLock_Entry &candidate) const
292 -> Dmn_DLock_ResultCode {
293 if (!candidate.isValid() || !candidate.m_range.isValid()) {
294 return Dmn_DLock_ResultCode::kInvalidRange;
295 }
296
297 for (const auto &entry : m_entries) {
298 if (entry.m_request_id == candidate.m_request_id &&
299 entry.m_session_id == candidate.m_session_id) {
300 continue;
301 }
302 if (entry.m_granted && candidate.m_granted &&
303 entry.m_domain == candidate.m_domain &&
304 entry.m_range.overlaps(candidate.m_range)) {
305 return Dmn_DLock_ResultCode::kConflict;
306 }
307 }
308
309 return Dmn_DLock_ResultCode::kOk;
310 }

313 return classifyGrantCandidate(candidate) == Dmn_DLock_ResultCode::kOk;

316 auto withCandidate(const Dmn_DLock_Entry &candidate) const
317 -> Dmn_DLock_TableSnapshot {
318 Dmn_DLock_TableSnapshot result = *this;
319 if (classifyGrantCandidate(candidate) != Dmn_DLock_ResultCode::kOk) {
320 return {};

324 result.m_table_version += 1;

584 auto acquireLock(Dmn_DLock_Range range,
585 const Dmn_DLock_RequestOptions &options = {})
586 -> Dmn_DLock_Result {
587 if (!range.isValid()) {
588 return {Dmn_DLock_ResultCode::kInvalidRange, range, options.m_request_id,
589 "invalid range"};
590 }
591
592 std::unique_lock<std::mutex> lock(m_mutex);
593 std::string request_id = options.m_request_id;
594 if (request_id.empty()) {
595 request_id =
596 std::string("req-") + std::to_string(m_table.m_next_sequence + 1);
597 }
598
599 while (m_table.isGrantedForRange(range)) {
600 if (!options.m_wait || options.m_no_wait) {
601 return {Dmn_DLock_ResultCode::kNoWait, range, request_id,
602 "range is already granted"};
603 }
604 m_cv.wait(lock);
605 }
606
607 Dmn_DLock_Entry entry{};
608 entry.m_domain = m_config.m_domain;
609 entry.m_request_id = request_id;
610 entry.m_session_id = std::string("session-") + request_id;
611 entry.m_owner_id = request_id;
612 entry.m_range = range;
613 entry.m_priority = 1;
614 entry.m_sequence = m_table.m_next_sequence + 1;
615 entry.m_fence = m_table.m_next_fencing_token + 1;
616 entry.m_state = Dmn_DLock_EntryState::kGranted;
617 entry.m_terminal_reason = Dmn_DLock_TerminalReason::kNone;
618 entry.m_acquire_deadline_ticks = 0;
619 entry.m_lease_deadline_ticks = options.m_lease_ticks;
620 entry.m_waiting = false;
621 entry.m_granted = true;
622
623 const auto candidate = m_table.withCandidate(entry);
624 if (candidate.m_entries.empty()) {
625 return {Dmn_DLock_ResultCode::kConflict, range, request_id,
626 "range already unavailable"};
627 }
628
629 m_table = candidate;
630 m_table.m_next_sequence = entry.m_sequence;
631 m_table.m_next_fencing_token = entry.m_fence;
632 m_cv.notify_all();
633 return {Dmn_DLock_ResultCode::kOk, range, request_id, "granted"};
634 }

636 auto acquireLockAsync(Dmn_DLock_Range range,
637 const Dmn_DLock_RequestOptions &options = {})
638 -> std::future<Dmn_DLock_Result> {
639 auto promise = std::make_shared<std::promise<Dmn_DLock_Result>>();
640 auto future = promise->get_future();
641 std::thread([this, range, options, promise]() {
642 auto result = acquireLock(range, options);
643 promise->set_value(result);
644 }).detach();
645 return future;

648 auto releaseLock(std::string_view request_id) -> Dmn_DLock_Result {
649 std::lock_guard<std::mutex> lock(m_mutex);
650 auto it = std::find_if(m_table.m_entries.begin(), m_table.m_entries.end(),
651 [&](const Dmn_DLock_Entry &entry) {
652 return entry.m_request_id == request_id;
653 });
654 if (it == m_table.m_entries.end()) {
655 return {Dmn_DLock_ResultCode::kInvalidState,
656 {},
657 std::string(request_id),
658 "request not found"};
659 }
660
661 const auto range = it->m_range;
662 m_table.m_entries.erase(it);
663 m_cv.notify_all();
664 return {Dmn_DLock_ResultCode::kOk, range, std::string(request_id),
665 "released"};
666 }
