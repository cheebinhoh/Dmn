262 auto canonicalize() -> void {
263 std::sort(m_entries.begin(), m_entries.end(),
264 [](const Dmn_DLock_Entry &lhs, const Dmn_DLock_Entry &rhs) {

272 auto validate() const -> bool {
273 if (m_domain.empty()) {
274 return false;
275 }
276
277 for (const auto &entry : m_entries) {
278 if (!entry.isValid() || entry.m_domain != m_domain) {
279 return false;
280 }
281
282 if (!entry.m_range.isValid()) {
283 return false;
284 }
285 }
286
287 for (std::size_t i = 0; i < m_entries.size(); ++i) {
288 for (std::size_t j = i + 1; j < m_entries.size(); ++j) {
289 const auto &lhs = m_entries[i];
290 const auto &rhs = m_entries[j];
291 const bool same_domain = lhs.m_domain == rhs.m_domain;
292 const bool granted_overlap =
293 lhs.m_granted && rhs.m_granted && lhs.m_range.overlaps(rhs.m_range);
294
295 if (same_domain && granted_overlap) {
296 return false;
297 }
298 }
299 }

301 return true;

304 auto classifyGrantCandidate(const Dmn_DLock_Entry &candidate) const
305 -> Dmn_DLock_ResultCode {
306 if (!candidate.isValid() || !candidate.m_range.isValid()) {
307 return Dmn_DLock_ResultCode::kInvalidRange;
308 }
309
310 if (candidate.m_domain != m_domain) {
311 return Dmn_DLock_ResultCode::kInvalidState;
312 }
313
314 for (const auto &entry : m_entries) {
315 if (entry.m_request_id == candidate.m_request_id &&
316 entry.m_session_id == candidate.m_session_id) {
317 continue;
318 }
319
320 if (entry.m_granted && candidate.m_granted &&
321 entry.m_domain == candidate.m_domain &&
322 entry.m_range.overlaps(candidate.m_range)) {
323 return Dmn_DLock_ResultCode::kConflict;
324 }
325 }
326
327 return Dmn_DLock_ResultCode::kOk;
328 }

330 auto canAcceptEntry(const Dmn_DLock_Entry &candidate) const -> bool {
331 return classifyGrantCandidate(candidate) == Dmn_DLock_ResultCode::kOk;

334 auto withCandidate(const Dmn_DLock_Entry &candidate) const
335 -> Dmn_DLock_TableSnapshot {

625 auto acquireLock(Dmn_DLock_Range range,
626 const Dmn_DLock_RequestOptions &options = {})
627 -> Dmn_DLock_Result {
628 if (!range.isValid()) {
629 return {Dmn_DLock_ResultCode::kInvalidRange, range, options.m_request_id,
630 "invalid range"};
631 }
632
633 std::unique_lock<std::mutex> lock(m_mutex);
634 std::string request_id = options.m_request_id;
635 if (request_id.empty()) {
636 request_id =
637 std::string("req-") + std::to_string(m_table.m_next_sequence + 1);
638 }
639
640 while (m_table.isGrantedForRange(range)) {
641 if (m_shutting_down.load(std::memory_order_acquire)) {
642 return {Dmn_DLock_ResultCode::kShutdown, range, request_id,
643 "lock is shutting down"};
644 }
645
646 if (!options.m_wait || options.m_no_wait) {
647 return {Dmn_DLock_ResultCode::kNoWait, range, request_id,
648 "range is already granted"};
649 }
650
651 m_cv.wait(lock);
652 }
653
654 if (m_shutting_down.load(std::memory_order_acquire)) {
655 return {Dmn_DLock_ResultCode::kShutdown, range, request_id,
656 "lock is shutting down"};
657 }
658
659 Dmn_DLock_Entry entry{};
660
661 entry.m_domain = m_config.m_domain;
662 entry.m_request_id = request_id;
663 entry.m_session_id = std::string("session-") + request_id;
664 entry.m_owner_id = request_id;
665 entry.m_range = range;
666 entry.m_priority = 1;
667 entry.m_sequence = m_table.m_next_sequence + 1;
668 entry.m_fence = m_table.m_next_fencing_token + 1;
669 entry.m_state = Dmn_DLock_EntryState::kGranted;
670 entry.m_terminal_reason = Dmn_DLock_TerminalReason::kNone;
671 entry.m_acquire_deadline_ticks = 0;
672 entry.m_lease_deadline_ticks = options.m_lease_ticks;
673 entry.m_waiting = false;
674 entry.m_granted = true;
675
676 const auto candidate = m_table.withCandidate(entry);
677 if (candidate.m_entries.empty()) {
678 return {Dmn_DLock_ResultCode::kConflict, range, request_id,
679 "range already unavailable"};
680 }
681
682 m_table = candidate;
683 m_table.m_next_sequence = entry.m_sequence;
684 m_table.m_next_fencing_token = entry.m_fence;
685 m_cv.notify_all();
686
687 return {Dmn_DLock_ResultCode::kOk, range, request_id, "granted"};
688 }

690 auto acquireLockAsync(Dmn_DLock_Range range,
691 const Dmn_DLock_RequestOptions &options = {})
692 -> std::future<Dmn_DLock_Result> {
693 std::lock_guard workersLock{m_async_workers_mutex};
694 if (m_shutting_down.load(std::memory_order_acquire)) {
695 std::promise<Dmn_DLock_Result> promise;
696 auto future = promise.get_future();
697 promise.set_value({Dmn_DLock_ResultCode::kShutdown, range,
698 options.m_request_id, "lock is shutting down"});
699
700 return future;
701 }
702
703 reapCompletedAsyncWorkers();
704
705 auto promise = std::make_shared<std::promise<Dmn_DLock_Result>>();
706 auto future = promise->get_future();
707 auto completion = std::make_shared<std::promise<void>>();
708 auto completionFuture = completion->get_future().share();
709 auto worker = std::make_unique<Dmn_Proc>(
710 "dlock-acquire", [this, range, options, promise, completion]() {
711 try {
712 promise->set_value(acquireLock(range, options));
713 } catch (...) {
714 promise->set_exception(std::current_exception());
715 }
716
717 completion->set_value();
718 });
719
720 auto *workerPtr = worker.get();
721 m_async_workers.push_back({std::move(worker), std::move(completionFuture)});
722 if (!workerPtr->exec()) {
723 m_async_workers.pop_back();
724 promise->set_exception(std::make_exception_ptr(std::runtime_error(
725 "failed to start Dmn_Proc worker for async lock acquisition")));
726 }
727
728 return future;
729 }

731 auto releaseLock(std::string_view request_id) -> Dmn_DLock_Result {
732 std::lock_guard<std::mutex> lock(m_mutex);
733
734 auto it = std::find_if(m_table.m_entries.begin(), m_table.m_entries.end(),
735 [&](const Dmn_DLock_Entry &entry) {
736 return entry.m_request_id == request_id;
737 });
738 if (it == m_table.m_entries.end()) {
739 return {Dmn_DLock_ResultCode::kInvalidState,
740 {},
741 std::string(request_id),
742 "request not found"};
743 }
744
745 const auto range = it->m_range;
746 m_table.m_entries.erase(it);
747 m_cv.notify_all();
748
749 return {Dmn_DLock_ResultCode::kOk, range, std::string(request_id),
750 "released"};
751 }

765 void reapCompletedAsyncWorkers() {
766 auto worker = m_async_workers.begin();
767 while (worker != m_async_workers.end()) {
768 if (worker->completion.wait_for(std::chrono::seconds::zero()) ==
769 std::future_status::ready) {
770 worker->process->wait();
