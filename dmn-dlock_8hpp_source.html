257 auto canonicalize() -> void {
258 std::sort(m_entries.begin(), m_entries.end(),
259 [](const Dmn_DLock_Entry &lhs, const Dmn_DLock_Entry &rhs) {
260 const auto lhs_key = lhs.canonicalKey();
261 const auto rhs_key = rhs.canonicalKey();

267 auto validate() const -> bool {
268 if (m_domain.empty()) {
269 return false;
270 }
271
272 for (const auto &entry : m_entries) {
273 if (!entry.isValid() || entry.m_domain != m_domain) {
274 return false;
275 }
276
277 if (!entry.m_range.isValid()) {
278 return false;
279 }
280 }
281
282 for (std::size_t i = 0; i < m_entries.size(); ++i) {
283 for (std::size_t j = i + 1; j < m_entries.size(); ++j) {
284 const auto &lhs = m_entries[i];
285 const auto &rhs = m_entries[j];
286 const bool same_domain = lhs.m_domain == rhs.m_domain;
287 const bool granted_overlap =
288 lhs.m_granted && rhs.m_granted && lhs.m_range.overlaps(rhs.m_range);
289
290 if (same_domain && granted_overlap) {
291 return false;
292 }
293 }
294 }
295
296 return true;

299 auto classifyGrantCandidate(const Dmn_DLock_Entry &candidate) const
300 -> Dmn_DLock_ResultCode {
301 if (!candidate.isValid() || !candidate.m_range.isValid()) {
302 return Dmn_DLock_ResultCode::kInvalidRange;
303 }
304
305 if (candidate.m_domain != m_domain) {
306 return Dmn_DLock_ResultCode::kInvalidState;
307 }
308
309 for (const auto &entry : m_entries) {
310 if (entry.m_request_id == candidate.m_request_id &&
311 entry.m_session_id == candidate.m_session_id) {
312 continue;
313 }
314
315 if (entry.m_granted && candidate.m_granted &&
316 entry.m_domain == candidate.m_domain &&
317 entry.m_range.overlaps(candidate.m_range)) {
318 return Dmn_DLock_ResultCode::kConflict;
319 }
320 }
321
322 return Dmn_DLock_ResultCode::kOk;
323 }

329 auto withCandidate(const Dmn_DLock_Entry &candidate) const
330 -> Dmn_DLock_TableSnapshot {

605 auto acquireLock(Dmn_DLock_Range range,
606 const Dmn_DLock_RequestOptions &options = {})
607 -> Dmn_DLock_Result {
608 if (!range.isValid()) {
609 return {Dmn_DLock_ResultCode::kInvalidRange, range, options.m_request_id,
610 "invalid range"};
611 }
612
613 std::unique_lock<std::mutex> lock(m_mutex);
614 std::string request_id = options.m_request_id;
615 if (request_id.empty()) {
616 request_id =
617 std::string("req-") + std::to_string(m_table.m_next_sequence + 1);
618 }
619
620 while (m_table.isGrantedForRange(range)) {
621 if (!options.m_wait || options.m_no_wait) {
622 return {Dmn_DLock_ResultCode::kNoWait, range, request_id,
623 "range is already granted"};
624 }
625
626 m_cv.wait(lock);
627 }
628
629 Dmn_DLock_Entry entry{};
630
631 entry.m_domain = m_config.m_domain;
632 entry.m_request_id = request_id;
633 entry.m_session_id = std::string("session-") + request_id;
634 entry.m_owner_id = request_id;
635 entry.m_range = range;
636 entry.m_priority = 1;
637 entry.m_sequence = m_table.m_next_sequence + 1;
638 entry.m_fence = m_table.m_next_fencing_token + 1;
639 entry.m_state = Dmn_DLock_EntryState::kGranted;
640 entry.m_terminal_reason = Dmn_DLock_TerminalReason::kNone;
641 entry.m_acquire_deadline_ticks = 0;
642 entry.m_lease_deadline_ticks = options.m_lease_ticks;
643 entry.m_waiting = false;
644 entry.m_granted = true;
645
646 const auto candidate = m_table.withCandidate(entry);
647 if (candidate.m_entries.empty()) {
648 return {Dmn_DLock_ResultCode::kConflict, range, request_id,
649 "range already unavailable"};
650 }
651
652 m_table = candidate;
653 m_table.m_next_sequence = entry.m_sequence;
654 m_table.m_next_fencing_token = entry.m_fence;
655 m_cv.notify_all();
656
657 return {Dmn_DLock_ResultCode::kOk, range, request_id, "granted"};
658 }

660 auto acquireLockAsync(Dmn_DLock_Range range,
661 const Dmn_DLock_RequestOptions &options = {})

674 auto releaseLock(std::string_view request_id) -> Dmn_DLock_Result {
675 std::lock_guard<std::mutex> lock(m_mutex);
676
677 auto it = std::find_if(m_table.m_entries.begin(), m_table.m_entries.end(),
678 [&](const Dmn_DLock_Entry &entry) {
679 return entry.m_request_id == request_id;
680 });
681 if (it == m_table.m_entries.end()) {
682 return {Dmn_DLock_ResultCode::kInvalidState,
683 {},
684 std::string(request_id),
685 "request not found"};
686 }
687
688 const auto range = it->m_range;
689 m_table.m_entries.erase(it);
690 m_cv.notify_all();
691
692 return {Dmn_DLock_ResultCode::kOk, range, std::string(request_id),
693 "released"};
