534 static auto classifyTopology(range_type range,
535 const std::vector<range_type> &overlaps)
536 -> Dmn_OverlayTopology {
537 if (overlaps.empty()) {
538 return Dmn_OverlayTopology::Clear;
539 }
540
541 std::vector<range_type> clippedRanges;
542 clippedRanges.reserve(overlaps.size());
543
544 for (const auto &overlap : overlaps) {
545 clippedRanges.push_back({std::max(range.m_start, overlap.m_start),
546 std::min(range.m_end, overlap.m_end)});
547 }
548
549 std::sort(clippedRanges.begin(), clippedRanges.end(),
550 [](const range_type &lhs, const range_type &rhs) {
551 if (lhs.m_start != rhs.m_start) {
552 return lhs.m_start < rhs.m_start;
553 }
554
555 return lhs.m_end < rhs.m_end;
556 });
557
558 bool leftCovered = clippedRanges.front().m_start == range.m_start;
559 bool rightCovered = false;
560 auto coveredEnd = clippedRanges.front().m_end;
561 bool continuousCoverage = leftCovered;
562
563 for (std::size_t index = 1; index < clippedRanges.size(); ++index) {
564 const auto &next = clippedRanges[index];
565
566 if (coveredEnd != std::numeric_limits<std::int64_t>::max() &&
567 next.m_start > coveredEnd + 1) {
568 continuousCoverage = false;
569 }
570
571 coveredEnd = std::max(coveredEnd, next.m_end);
572 }
573
574 rightCovered = coveredEnd == range.m_end;
575 if (continuousCoverage && rightCovered) {
576 return Dmn_OverlayTopology::FullyCovered;
577 }
578
579 auto overlapStart = overlaps.front().m_start;
580 auto overlapEnd = overlaps.front().m_end;
581 bool hasContainedRange = false;
582
583 for (const auto &overlap : overlaps) {
584 overlapStart = std::min(overlapStart, overlap.m_start);
585 overlapEnd = std::max(overlapEnd, overlap.m_end);
586 hasContainedRange =
587 hasContainedRange ||
588 (range.m_start < overlap.m_start && overlap.m_end < range.m_end);
589 }
590
591 if (hasContainedRange && range.m_start < overlapStart &&
592 range.m_end > overlapEnd) {
593 return Dmn_OverlayTopology::CoveringExisting;
594 }
595
596 if (leftCovered && rightCovered) {
597 return Dmn_OverlayTopology::OverlaidBoth;
598 }
599
600 if (leftCovered) {
601 return Dmn_OverlayTopology::OverlaidLeft;
602 }
603
604 if (rightCovered) {
605 return Dmn_OverlayTopology::OverlaidRight;
606 }
607
608 return Dmn_OverlayTopology::Clear;
609 }

645 auto entryLess(const Entry &lhs, const Entry &rhs) const -> bool {
646 const bool identicalRanges = lhs.m_range.m_start == rhs.m_range.m_start &&
647 lhs.m_range.m_end == rhs.m_range.m_end;
648
649 if (identicalRanges) {

714 for (const Entry *entry : entries) {

716 overlaps.push_back(entry->m_range);

723 }
724 }

726 state.m_topology = classifyTopology(candidate->m_range, overlaps);

801 transitions;

805 entry->m_state = newState;

811 }

814 return;

818 return;

824 for (const auto &[entry, oldState, newState] : transitions) {
825 if (entry->m_onStateChange) {

1004 void eraseEntry(Node &node, const Entry *target) {
1005 std::size_t index = 0;
1006 while (index < node.m_entries.size() &&
1007 entryLess(*node.m_entries[index], *target)) {
1008 ++index;
1009 }
1010
1011 if (index < node.m_entries.size() &&
1012 node.m_entries[index].get() == target) {
1013 if (node.m_isLeaf) {
1014 node.m_entries.erase(node.m_entries.begin() +
1015 static_cast<std::ptrdiff_t>(index));
1016 refreshNodeMaxEnd(&node);
1017 } else if (node.m_children[index]->m_entries.size() >= kMinimumDegree) {
1018 node.m_entries[index] = removeMaximum(*node.m_children[index]);
1019 refreshNodeMaxEnd(&node);

1021 kMinimumDegree) {
1022 node.m_entries[index] = removeMinimum(*node.m_children[index + 1]);
1023 refreshNodeMaxEnd(&node);

1026 eraseEntry(*node.m_children[index], target);
1027 refreshNodeMaxEnd(&node);

1032
1033 if (node.m_isLeaf) {
1034 return;
1035 }

1038 if (index > 0 &&
