503 static auto classifyTopology(range_type range,
504 const std::vector<range_type> &overlaps)
505 -> Dmn_OverlayTopology {
506 if (overlaps.empty()) {
507 return Dmn_OverlayTopology::Clear;
508 }
509
510 std::vector<range_type> clippedRanges;
511 clippedRanges.reserve(overlaps.size());
512 for (const auto &overlap : overlaps) {
513 clippedRanges.push_back({std::max(range.m_start, overlap.m_start),
514 std::min(range.m_end, overlap.m_end)});
515 }
516 std::sort(clippedRanges.begin(), clippedRanges.end(),
517 [](const range_type &lhs, const range_type &rhs) {
518 if (lhs.m_start != rhs.m_start) {
519 return lhs.m_start < rhs.m_start;
520 }
521 return lhs.m_end < rhs.m_end;
522 });
523
524 bool leftCovered = clippedRanges.front().m_start == range.m_start;
525 bool rightCovered = false;
526 auto coveredEnd = clippedRanges.front().m_end;
527 bool continuousCoverage = leftCovered;
528 for (std::size_t index = 1; index < clippedRanges.size(); ++index) {
529 const auto &next = clippedRanges[index];
530 if (coveredEnd != std::numeric_limits<std::int64_t>::max() &&
531 next.m_start > coveredEnd + 1) {
532 continuousCoverage = false;
533 }
534 coveredEnd = std::max(coveredEnd, next.m_end);
535 }
536 rightCovered = coveredEnd == range.m_end;
537 if (continuousCoverage && rightCovered) {
538 return Dmn_OverlayTopology::FullyCovered;
539 }
540
541 auto overlapStart = overlaps.front().m_start;
542 auto overlapEnd = overlaps.front().m_end;
543 bool hasContainedRange = false;
544 for (const auto &overlap : overlaps) {
545 overlapStart = std::min(overlapStart, overlap.m_start);
546 overlapEnd = std::max(overlapEnd, overlap.m_end);
547 hasContainedRange =
548 hasContainedRange ||
549 (range.m_start < overlap.m_start && overlap.m_end < range.m_end);
550 }
551 if (hasContainedRange && range.m_start < overlapStart &&
552 range.m_end > overlapEnd) {
553 return Dmn_OverlayTopology::CoveringExisting;
554 }
555 if (leftCovered && rightCovered) {
556 return Dmn_OverlayTopology::OverlaidBoth;
557 }
558 if (leftCovered) {
559 return Dmn_OverlayTopology::OverlaidLeft;
560 }
561 if (rightCovered) {
562 return Dmn_OverlayTopology::OverlaidRight;
563 }
564 return Dmn_OverlayTopology::Clear;
565 }

600 auto entryLess(const Entry &lhs, const Entry &rhs) const -> bool {
601 const bool identicalRanges = lhs.m_range.m_start == rhs.m_range.m_start &&
602 lhs.m_range.m_end == rhs.m_range.m_end;
603 if (identicalRanges) {

649 auto calculateState(const Entry *candidate) const -> Dmn_OverlayState {
650 Dmn_OverlayState state;
651 std::vector<range_type> overlaps;
652 std::vector<const Entry *> entries;
653 entries.reserve(m_size);
654 if (m_root) {
655 collectEntriesInOrder(m_root.get(), entries);
656 }
657 for (const Entry *entry : entries) {
658 if (entry != candidate && candidate->m_range.overlaps(entry->m_range)) {
659 overlaps.push_back(entry->m_range);
660 if (m_priorityEvaluator) {
661 MutationBlockGuard guard(m_dispatchingCallbacks);
662 if (m_priorityEvaluator(entry->m_value, candidate->m_value)) {
663 state.m_isTop = false;
664 }
665 }
666 }
667 }
668 state.m_topology = classifyTopology(candidate->m_range, overlaps);
669 return state;
670 }

691 existingEntries.reserve(m_size);

693 collectEntriesInOrder(m_root.get(), existingEntries);
694 }

696 if (existing->m_range.overlaps(entry->m_range)) {
697 oldStates.emplace_back(existing, existing->m_state);
698 }
699 }

701 auto *insertedEntry = entry.get();
702 if (!m_root) {
703 m_root = std::make_unique<Node>();

706 auto newRoot = std::make_unique<Node>();

716 m_root = std::move(newRoot);

719 insertNonFull(*m_root, std::move(entry));
720 ++m_nextOrdinal;
721 ++m_size;

723 insertedEntry->m_state = calculateState(insertedEntry);
724 updateStatesAndNotify(oldStates);

728 void updateStatesAndNotify(
729 const std::vector<std::pair<Entry *, Dmn_OverlayState>> &oldStates) {
730 std::vector<std::tuple<Entry *, Dmn_OverlayState, Dmn_OverlayState>>
731 transitions;
732 for (const auto &[entry, oldState] : oldStates) {
733 const auto newState = calculateState(entry);
734 entry->m_state = newState;
735 if (oldState.m_topology != newState.m_topology ||
736 oldState.m_isTop != newState.m_isTop) {
737 transitions.emplace_back(entry, oldState, newState);
738 }
739 }
740 if (transitions.empty()) {
741 return;
742 }
743 if (m_suppressCallbacks) {
744 return;
745 }
746
747 {
748 MutationBlockGuard guard(m_dispatchingCallbacks);
749 for (const auto &[entry, oldState, newState] : transitions) {
750 if (entry->m_onStateChange) {
751 entry->m_onStateChange(entry->m_range, entry->m_value, oldState,
752 newState);
753 }
754 }
755 }

907 void eraseEntry(Node &node, const Entry *target) {
908 std::size_t index = 0;
909 while (index < node.m_entries.size() &&
910 entryLess(*node.m_entries[index], *target)) {
911 ++index;
912 }
913
914 if (index < node.m_entries.size() &&
915 node.m_entries[index].get() == target) {
916 if (node.m_isLeaf) {
917 node.m_entries.erase(node.m_entries.begin() +
918 static_cast<std::ptrdiff_t>(index));
919 refreshNodeMaxEnd(&node);
920 } else if (node.m_children[index]->m_entries.size() >= kMinimumDegree) {
921 node.m_entries[index] = removeMaximum(*node.m_children[index]);
922 refreshNodeMaxEnd(&node);
923 } else if (node.m_children[index + 1]->m_entries.size() >=
924 kMinimumDegree) {
925 node.m_entries[index] = removeMinimum(*node.m_children[index + 1]);
926 refreshNodeMaxEnd(&node);
927 } else {
928 mergeChildren(node, index);
929 eraseEntry(*node.m_children[index], target);
930 refreshNodeMaxEnd(&node);
931 }
932 return;
933 }
934 if (node.m_isLeaf) {
935 return;
936 }
937
938 if (node.m_children[index]->m_entries.size() == kMinimumDegree - 1) {
939 if (index > 0 &&
940 node.m_children[index - 1]->m_entries.size() >= kMinimumDegree) {
941 borrowFromPrevious(node, index);
942 } else if (index < node.m_entries.size() &&
943 node.m_children[index + 1]->m_entries.size() >=
944 kMinimumDegree) {
945 borrowFromNext(node, index);
946 } else if (index < node.m_entries.size()) {
947 mergeChildren(node, index);
948 } else {
949 mergeChildren(node, index - 1);
950 --index;
951 }
952 }
953 eraseEntry(*node.m_children[index], target);
954 refreshNodeMaxEnd(&node);
955 }

979 void splitChild(Node &parent, std::size_t childIndex) {
980 Node &child = *parent.m_children[childIndex];
981 auto sibling = std::make_unique<Node>();
982 sibling->m_isLeaf = child.m_isLeaf;
983
984 constexpr auto maxKeys = 2 * kMinimumDegree - 1;
985 sibling->m_entries.reserve(kMinimumDegree - 1);
986 if (!child.m_isLeaf) {
987 sibling->m_children.reserve(kMinimumDegree);
988 }
989 parent.m_entries.reserve(parent.m_entries.size() + 1);
990 parent.m_children.reserve(parent.m_children.size() + 1);
991
992 for (std::size_t index = kMinimumDegree; index < maxKeys; ++index) {
993 sibling->m_entries.push_back(std::move(child.m_entries[index]));
994 }
995 auto promoted = std::move(child.m_entries[kMinimumDegree - 1]);
996 child.m_entries.resize(kMinimumDegree - 1);
997
998 if (!child.m_isLeaf) {
999 for (std::size_t index = kMinimumDegree; index < child.m_children.size();
1000 ++index) {
1001 sibling->m_children.push_back(std::move(child.m_children[index]));
1002 }
1003 child.m_children.resize(kMinimumDegree);
1004 }
1005
1006 parent.m_entries.insert(parent.m_entries.begin() +
1007 static_cast<std::ptrdiff_t>(childIndex),
1008 std::move(promoted));
1009 parent.m_children.insert(parent.m_children.begin() +
1010 static_cast<std::ptrdiff_t>(childIndex + 1),
1011 std::move(sibling));
1012 parent.m_isLeaf = false;
1013 refreshNodeMaxEnd(&child);
1014 refreshNodeMaxEnd(parent.m_children[childIndex + 1].get());
1015 refreshNodeMaxEnd(&parent);
1016 }

1018 void insertNonFull(Node &node, std::unique_ptr<Entry> entry) {
1019 auto childIndex = node.m_entries.size();
1020 while (childIndex > 0 &&
1021 entryLess(*entry, *node.m_entries[childIndex - 1])) {
1022 --childIndex;
1023 }
1024
1025 if (node.m_isLeaf) {
1026 node.m_entries.reserve(node.m_entries.size() + 1);
1027 node.m_entries.insert(node.m_entries.begin() +
1028 static_cast<std::ptrdiff_t>(childIndex),
1029 std::move(entry));
1030 refreshNodeMaxEnd(&node);
1031 return;
1032 }
1033
1034 if (node.m_children[childIndex]->m_entries.size() ==
1035 2 * kMinimumDegree - 1) {
1036 splitChild(node, childIndex);
1037 if (entryLess(*node.m_entries[childIndex], *entry)) {
1038 ++childIndex;
1039 }
1040 }
1041
1042 insertNonFull(*node.m_children[childIndex], std::move(entry));
1043 refreshNodeMaxEnd(&node);
1044 }

1046 auto hasOverlapInSubtree(const Node *node,
1047 range_type range) const noexcept -> bool {
1048 if (node->m_subtreeMaxEnd < range.m_start) {
1049 return false;
1050 }
1051 for (std::size_t index = 0; index < node->m_entries.size(); ++index) {
1052 if (!node->m_isLeaf &&
1053 hasOverlapInSubtree(node->m_children[index].get(), range)) {
1054 return true;
1055 }
1056 if (node->m_entries[index]->m_range.overlaps(range)) {
1057 return true;
1058 }
1059 }
1060 return !node->m_isLeaf &&
1061 hasOverlapInSubtree(node->m_children.back().get(), range);
1062 }

1064 void visitOverlappingInOrder(const Node *node, range_type range,
1065 const overlap_visitor &visitor) const {

1068 }

1071 visitOverlappingInOrder(node->m_children[index].get(), range, visitor);

1074 if (entry.m_range.overlaps(range)) {
1075 visitor(entry.m_range, entry.m_value);
1076 }
1077 }
1078 if (!node->m_isLeaf) {
1079 visitOverlappingInOrder(node->m_children.back().get(), range, visitor);

1182 entries.reserve(m_size);
1183 collectEntriesInOrder(m_root.get(), entries);
1184 std::vector<Entry *> targets;
1185 for (Entry *entry : entries) {

1200 targets.begin(), targets.end(), [entry](const Entry *target) {

1202 });

1204 oldStates.emplace_back(entry, entry->m_state);
1205 }
1206 }
1207
1208 for (Entry *target : targets) {
1209 eraseEntry(target);
1210 }
