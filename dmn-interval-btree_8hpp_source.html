551 static auto classifyTopology(range_type range,
552 const std::vector<range_type> &overlaps)
553 -> Dmn_OverlayTopology {
554 if (overlaps.empty()) {
555 return Dmn_OverlayTopology::Clear;
556 }
557
558 std::vector<range_type> clippedRanges;
559 clippedRanges.reserve(overlaps.size());
560
561 for (const auto &overlap : overlaps) {
562 clippedRanges.push_back({std::max(range.m_start, overlap.m_start),
563 std::min(range.m_end, overlap.m_end)});
564 }
565
566 std::sort(clippedRanges.begin(), clippedRanges.end(),
567 [](const range_type &lhs, const range_type &rhs) {
568 if (lhs.m_start != rhs.m_start) {
569 return lhs.m_start < rhs.m_start;
570 }
571
572 return lhs.m_end < rhs.m_end;
573 });
574
575 bool leftCovered = clippedRanges.front().m_start == range.m_start;
576 bool rightCovered = false;
577 auto coveredEnd = clippedRanges.front().m_end;
578 bool continuousCoverage = leftCovered;
579
580 for (std::size_t index = 1; index < clippedRanges.size(); ++index) {
581 const auto &next = clippedRanges[index];
582
583 if (coveredEnd != std::numeric_limits<std::int64_t>::max() &&
584 next.m_start > coveredEnd + 1) {
585 continuousCoverage = false;
586 }
587
588 coveredEnd = std::max(coveredEnd, next.m_end);
589 }
590
591 rightCovered = coveredEnd == range.m_end;
592 if (continuousCoverage && rightCovered) {
593 return Dmn_OverlayTopology::FullyCovered;
594 }
595
596 auto overlapStart = overlaps.front().m_start;
597 auto overlapEnd = overlaps.front().m_end;
598 bool hasContainedRange = false;
599
600 for (const auto &overlap : overlaps) {
601 overlapStart = std::min(overlapStart, overlap.m_start);
602 overlapEnd = std::max(overlapEnd, overlap.m_end);
603 hasContainedRange =
604 hasContainedRange ||
605 (range.m_start < overlap.m_start && overlap.m_end < range.m_end);
606 }
607
608 if (hasContainedRange && range.m_start < overlapStart &&
609 range.m_end > overlapEnd) {
610 return Dmn_OverlayTopology::CoveringExisting;
611 }
612
613 if (leftCovered && rightCovered) {
614 return Dmn_OverlayTopology::OverlaidBoth;
615 }
616
617 if (leftCovered) {
618 return Dmn_OverlayTopology::OverlaidLeft;
619 }
620
621 if (rightCovered) {
622 return Dmn_OverlayTopology::OverlaidRight;
623 }
624
625 return Dmn_OverlayTopology::Clear;
626 }

662 auto entryLess(const Entry &lhs, const Entry &rhs) const -> bool {
663 const bool identicalRanges = lhs.m_range.m_start == rhs.m_range.m_start &&
664 lhs.m_range.m_end == rhs.m_range.m_end;
665
666 if (identicalRanges) {
667 return lhs.m_ordinal < rhs.m_ordinal;
668 }
669
670 if (m_canonicalComparator) {
671 MutationBlockGuard guard(m_dispatchingCallbacks);
672
673 if (m_canonicalComparator(lhs.m_range, rhs.m_range)) {
674 return true;
675 }
676
677 if (m_canonicalComparator(rhs.m_range, lhs.m_range)) {
678 return false;
679 }
680 }
681
682 return defaultRangeLess(lhs.m_range, rhs.m_range);
683 }

720 auto calculateState(const Entry *candidate) const -> Dmn_OverlayState {
721 Dmn_OverlayState state;
722
723 std::vector<range_type> overlaps;
724 std::vector<const Entry *> entries;
725 entries.reserve(m_size);
726
727 if (m_root) {
728 collectEntriesInOrder(m_root.get(), entries);
729 }
730
731 for (const Entry *entry : entries) {
732 if (entry != candidate && candidate->m_range.overlaps(entry->m_range)) {
733 overlaps.push_back(entry->m_range);
734 if (m_priorityEvaluator) {
735 MutationBlockGuard guard(m_dispatchingCallbacks);
736 if (m_priorityEvaluator(entry->m_value, candidate->m_value)) {
737 state.m_isTop = false;
738 }
739 }
740 }
741 }
742
743 state.m_topology = classifyTopology(candidate->m_range, overlaps);
744
745 return state;
746 }

763 auto insertEntry(std::unique_ptr<Entry> entry) -> bool {
764 if (!entry->m_range.isValid()) {
765 throw std::invalid_argument("range is invalid");
766 }
767
768 std::vector<Entry *> existingEntries;
769 std::vector<std::pair<Entry *, Dmn_OverlayState>> oldStates;
770 existingEntries.reserve(m_size);
771
772 if (m_root) {
773 collectEntriesInOrder(m_root.get(), existingEntries);
774 }
775
776 for (Entry *existing : existingEntries) {
777 if (existing->m_range.overlaps(entry->m_range)) {
778 oldStates.emplace_back(existing, existing->m_state);
779 }
780 }
781
782 auto *insertedEntry = entry.get();
783 if (!m_root) {
784 m_root = std::make_unique<Node>();
785 }
786
787 if (m_root->m_entries.size() == 2 * kMinimumDegree - 1) {
788 auto newRoot = std::make_unique<Node>();
789
790 newRoot->m_isLeaf = false;
791 newRoot->m_children.reserve(1);
792 newRoot->m_children.push_back(std::move(m_root));
793
794 try {
795 splitChild(*newRoot, 0);
796 } catch (...) {
797 m_root = std::move(newRoot->m_children.front());
798
799 throw;
800 }
801
802 m_root = std::move(newRoot);
803 }
804
805 insertNonFull(*m_root, std::move(entry));
806 ++m_nextOrdinal;
807 ++m_size;
808
809 insertedEntry->m_state = calculateState(insertedEntry);
810 updateStatesAndNotify(oldStates);
811
812 return true;
813 }

815 void updateStatesAndNotify(
816 const std::vector<std::pair<Entry *, Dmn_OverlayState>> &oldStates) {
817 std::vector<std::tuple<Entry *, Dmn_OverlayState, Dmn_OverlayState>>
818 transitions;
819
820 for (const auto &[entry, oldState] : oldStates) {
821 const auto newState = calculateState(entry);
822 entry->m_state = newState;
823
824 if (oldState.m_topology != newState.m_topology ||
825 oldState.m_isTop != newState.m_isTop) {
826 transitions.emplace_back(entry, oldState, newState);
827 }
828 }
829
830 if (transitions.empty()) {
831 return;
832 }
833
834 if (m_suppressCallbacks) {
835 return;
836 }
837
838 {
839 MutationBlockGuard guard(m_dispatchingCallbacks);
840
841 for (const auto &[entry, oldState, newState] : transitions) {
842 if (entry->m_onStateChange) {
843 entry->m_onStateChange(entry->m_range, entry->m_value, oldState,
844 newState);
845 }
846 }
847 }

913 if (node.m_isLeaf) {

999 void mergeChildren(Node &parent, std::size_t separatorIndex) {
1000 Node &left = *parent.m_children[separatorIndex];
1001 Node &right = *parent.m_children[separatorIndex + 1];
1002 left.m_entries.push_back(std::move(parent.m_entries[separatorIndex]));
1003 for (auto &entry : right.m_entries) {
1004 left.m_entries.push_back(std::move(entry));
1005 }

1008 for (auto &child : right.m_children) {
1009 left.m_children.push_back(std::move(child));

1011 }

1014 static_cast<std::ptrdiff_t>(separatorIndex));
1015 parent.m_children.erase(parent.m_children.begin() +
1016 static_cast<std::ptrdiff_t>(separatorIndex + 1));
1017 refreshNodeMaxEnd(&left);

1021 void eraseEntry(Node &node, const Entry *target) {
1022 std::size_t index = 0;
1023 while (index < node.m_entries.size() &&
1024 entryLess(*node.m_entries[index], *target)) {
1025 ++index;
1026 }
1027
1028 if (index < node.m_entries.size() &&
1029 node.m_entries[index].get() == target) {
1030 if (node.m_isLeaf) {
1031 node.m_entries.erase(node.m_entries.begin() +
1032 static_cast<std::ptrdiff_t>(index));
1033 refreshNodeMaxEnd(&node);
1034 } else if (node.m_children[index]->m_entries.size() >= kMinimumDegree) {
1035 node.m_entries[index] = removeMaximum(*node.m_children[index]);
1036 refreshNodeMaxEnd(&node);
1037 } else if (node.m_children[index + 1]->m_entries.size() >=
1038 kMinimumDegree) {
1039 node.m_entries[index] = removeMinimum(*node.m_children[index + 1]);
1040 refreshNodeMaxEnd(&node);
1041 } else {
1042 mergeChildren(node, index);
1043 eraseEntry(*node.m_children[index], target);
1044 refreshNodeMaxEnd(&node);
1045 }
1046
1047 return;
1048 }
1049
1050 if (node.m_isLeaf) {
1051 return;
1052 }
1053
1054 if (node.m_children[index]->m_entries.size() == kMinimumDegree - 1) {
1055 if (index > 0 &&
1056 node.m_children[index - 1]->m_entries.size() >= kMinimumDegree) {
1057 borrowFromPrevious(node, index);
1058 } else if (index < node.m_entries.size() &&
1059 node.m_children[index + 1]->m_entries.size() >=
1060 kMinimumDegree) {
1061 borrowFromNext(node, index);
1062 } else if (index < node.m_entries.size()) {
1063 mergeChildren(node, index);
1064 } else {
1065 mergeChildren(node, index - 1);
1066 --index;
1067 }
1068 }
1069
1070 eraseEntry(*node.m_children[index], target);
1071 refreshNodeMaxEnd(&node);
1072 }

1099 void splitChild(Node &parent, std::size_t childIndex) {
1100 Node &child = *parent.m_children[childIndex];
1101 auto sibling = std::make_unique<Node>();
1102 sibling->m_isLeaf = child.m_isLeaf;
1103
1104 constexpr auto maxKeys = 2 * kMinimumDegree - 1;
1105 sibling->m_entries.reserve(kMinimumDegree - 1);
1106
1107 if (!child.m_isLeaf) {
1108 sibling->m_children.reserve(kMinimumDegree);
1109 }
1110
1111 parent.m_entries.reserve(parent.m_entries.size() + 1);
1112 parent.m_children.reserve(parent.m_children.size() + 1);
1113
1114 for (std::size_t index = kMinimumDegree; index < maxKeys; ++index) {
1115 sibling->m_entries.push_back(std::move(child.m_entries[index]));
1116 }
1117
1118 auto promoted = std::move(child.m_entries[kMinimumDegree - 1]);
1119 child.m_entries.resize(kMinimumDegree - 1);
1120
1121 if (!child.m_isLeaf) {
1122 for (std::size_t index = kMinimumDegree; index < child.m_children.size();
1123 ++index) {
1124 sibling->m_children.push_back(std::move(child.m_children[index]));
1125 }
1126
1127 child.m_children.resize(kMinimumDegree);
1128 }
1129
1130 parent.m_entries.insert(parent.m_entries.begin() +
1131 static_cast<std::ptrdiff_t>(childIndex),
1132 std::move(promoted));
1133 parent.m_children.insert(parent.m_children.begin() +
1134 static_cast<std::ptrdiff_t>(childIndex + 1),
1135 std::move(sibling));
1136 parent.m_isLeaf = false;
1137 refreshNodeMaxEnd(&child);
1138 refreshNodeMaxEnd(parent.m_children[childIndex + 1].get());
1139 refreshNodeMaxEnd(&parent);
1140 }

1142 void insertNonFull(Node &node, std::unique_ptr<Entry> entry) {
1143 auto childIndex = node.m_entries.size();
1144 while (childIndex > 0 &&
1145 entryLess(*entry, *node.m_entries[childIndex - 1])) {
1146 --childIndex;
1147 }
1148
1149 if (node.m_isLeaf) {
1150 node.m_entries.reserve(node.m_entries.size() + 1);
1151 node.m_entries.insert(node.m_entries.begin() +
1152 static_cast<std::ptrdiff_t>(childIndex),
1153 std::move(entry));
1154 refreshNodeMaxEnd(&node);
1155
1156 return;
1157 }
1158
1159 if (node.m_children[childIndex]->m_entries.size() ==
1160 2 * kMinimumDegree - 1) {
1161 splitChild(node, childIndex);
1162
1163 if (entryLess(*node.m_entries[childIndex], *entry)) {
1164 ++childIndex;
1165 }
1166 }
1167
1168 insertNonFull(*node.m_children[childIndex], std::move(entry));
1169 refreshNodeMaxEnd(&node);
1170 }

1172 auto hasOverlapInSubtree(const Node *node,
1173 range_type range) const noexcept -> bool {
1174 if (node->m_subtreeMaxEnd < range.m_start) {
1175 return false;
1176 }
1177
1178 for (std::size_t index = 0; index < node->m_entries.size(); ++index) {
1179 if (!node->m_isLeaf &&
1180 hasOverlapInSubtree(node->m_children[index].get(), range)) {
1181 return true;
1182 }
1183
1184 if (node->m_entries[index]->m_range.overlaps(range)) {
1185 return true;
1186 }
1187 }
1188
1189 return !node->m_isLeaf &&
1190 hasOverlapInSubtree(node->m_children.back().get(), range);
1191 }
