198 static_assert(std::numeric_limits<Rep>::radix == 2,

201 const long double exclusive_max =

214 static auto nextDeadline(Clock::duration interval) -> Clock::time_point {
215 using Rep = Clock::duration::rep;
216 const auto now = Clock::now();
217 const auto now_ticks = now.time_since_epoch().count();
218 const auto interval_ticks = interval.count();
219
220 if (now_ticks > 0 &&
221 interval_ticks > std::numeric_limits<Rep>::max() - now_ticks) {
222 throw std::overflow_error(
223 "Dmn_Timer deadline exceeds steady_clock range");
224 }
225
226 return Clock::time_point{Clock::duration{now_ticks + interval_ticks}};
