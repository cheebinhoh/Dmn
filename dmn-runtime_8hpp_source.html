487Dmn_Runtime_Manager<QueueType>::Dmn_Runtime_Manager()
488 : Dmn_Singleton<Dmn_Runtime_Manager<QueueType>>{},
489 Dmn_Async<QueueType>{"Dmn_Runtime_Manager"},
490 m_mask{Dmn_Runtime_Manager::s_mask} {
491 this->addExecTask([this]() { m_asyncThreadId = std::this_thread::get_id(); });
492
493 // default, these signal handler hooks will be executed in the singleton
494 // asynchronous context right after externally registered signal handler hooks
495 m_signal_handler_hooks[SIGTERM] = [this]([[maybe_unused]] int signo) -> void {
496 this->exitMainLoop();
497 };
498
499 m_signal_handler_hooks[SIGINT] = [this]([[maybe_unused]] int signo) -> void {
500 this->exitMainLoop();
501 };
502
503 m_signal_handler_hooks[SIGALRM] = [this]([[maybe_unused]] int signo) -> void {
504 assert(isRunInAsyncThread());
505
506 if (m_timedQueue.empty()) {
507 return;
508 }
509
510 const TimePoint &tp = std::chrono::steady_clock::now();
511 while (!m_timedQueue.empty()) {
512 const auto &job = m_timedQueue.top();
513 if (tp >= job.m_due) {
514 this->addJob(job.m_fnc, job.m_priority, job.m_onErrorFnc);
515
516 m_timedQueue.pop();
517 } else {
518 this->setNextTimerAt(job.m_due);

520 break;
521 }
522 }
523 };

526 m_pimpl = detail::Dmn_Runtime_Manager_Impl_create();

560 .m_fnc =

562 .m_onErrorFnc = std::move(onErrorFnc)};

566 m_highQueue.push(std::move(rjob));

697template <template <class> class QueueType>
698void Dmn_Runtime_Manager<QueueType>::execRuntimeJobInternal() {
699 Dmn_Runtime_Job::Priority state{}; // not high, not medium, not low
700
701 assert(isRunInAsyncThread());
702 assert(this->m_sched_job.size() == this->m_sched_task.size());
703
704 if (!this->m_sched_job.empty()) {
705 state = this->m_sched_job.top().m_priority;
706 }
707
708 try {
709 if (state != Dmn_Runtime_Job::Priority::kHigh) {
710 auto item = m_highQueue.popNoWait();
711
712 if (item) {
713 this->addRuntimeJobToCoroutineSchedulerContext(std::move(*item));
714 } else if (state != Dmn_Runtime_Job::Priority::kMedium) {
715 item = m_mediumQueue.popNoWait();
716
717 if (item) {
718 this->addRuntimeJobToCoroutineSchedulerContext(std::move(*item));
719 } else if (state != Dmn_Runtime_Job::Priority::kLow) {
720
721 item = m_lowQueue.popNoWait();
722 if (item) {
723 this->addRuntimeJobToCoroutineSchedulerContext(std::move(*item));
724 }
725 }
726 }
727 }
728
729 assert(this->m_sched_job.size() == this->m_sched_task.size());
730
731 auto jobsCountInScheduler = this->m_sched_job.size();
732
733 if (jobsCountInScheduler > 0) {
734 bool done = this->runRuntimeCoroutineScheduler();
735
736 if (!done) {
737 assert(!this->m_sched_job.empty());
738 assert(!this->m_sched_task.empty());
739 assert(this->m_sched_job.size() == jobsCountInScheduler);
740
741 this->runRuntimeJobExecutor();
742 } else if (this->m_sched_job.size() < jobsCountInScheduler &&
743 this->m_jobs_count.fetch_sub(1) > 1) {
744 this->runRuntimeJobExecutor();
745 } else {
746 assert(this->m_sched_job.empty());
747 assert(this->m_sched_task.empty());
748 assert(0 == m_jobs_count.load(std::memory_order_acquire));
749 }
750 }
751 } catch (...) {
752 if (this->m_jobs_count.fetch_sub(1) > 1) {
753 this->runRuntimeJobExecutor();
754 }
755 }
756}

762template <template <class> class QueueType>
763void Dmn_Runtime_Manager<QueueType>::exitMainLoop() {
764 if (!m_main_exit_flag.test_and_set(std::memory_order_release)) {
765 m_main_enter_flag.clear(std::memory_order_release);
766
767 // set the last timer, so that signalWaitProc gracefully exit.
768 this->setNextTimer(0, 10000);
769
770 // after setting the main loop enter (false) and exit (true) flags, we just
771 // have to wait for signalWaitProc to gracefully exit as it is triggered
772 // at fixed intervals before we clear the timer.
773 if (m_signalWaitProc) {
774 m_signalWaitProc->wait();
775 m_signalWaitProc = {};
776 }
777 }
778
779 if (m_pimpl) {
780 this->setNextTimer(
781 0, 0); // disable timer, though m_signalWaitProc should already have
782 // done so; if it has crashed, we still disable the timer here.
783 }
784
785 m_main_exit_flag.notify_all();
786}

796template <template <class> class QueueType>
797void Dmn_Runtime_Manager<QueueType>::enterMainLoop() {
798 // If there are already pending jobs (e.g., from a prior run), ensure the
799 // runtime job executor is started so the system does not remain idle.
800 bool startJobExecutor = m_jobs_count.load(std::memory_order_acquire) > 0;
801
802 if (m_main_enter_flag.test_and_set(std::memory_order_acquire)) {
803 assert(false && "Error: enter main loop twice without exit");
804
805 throw std::runtime_error("Error: enter main loop twice without exit");
806 }
807
808 m_main_exit_flag.clear(std::memory_order_release);
809 m_main_enter_flag.notify_all();
810 Dmn_Proc::yield();
811
812 m_signalWaitProc = std::make_unique<Dmn_Proc>(
813 "DmnRuntimeManager_SignalWait", [this]() -> void {
814 while (true) {
815 int signo{};
816 int err{};
817
818 err = sigwait(&m_mask, &signo);
819 if (err) {
820 throw std::runtime_error("Error in sigwait: " +
821 std::system_category().message(err));
822 }
823
824 if (m_main_exit_flag.test(std::memory_order_acquire)) {
825 break;
826 } else {
827 this->addExecTask([this, signo]() {
828 this->execSignalHandlerHookInternal(signo);
829 });
830 }
831 }
832 });
833
834 if (!m_signalWaitProc->exec()) {
835 throw std::runtime_error(
836 "Failed to start DmnRuntimeManager_SignalWait task");
837 }
838
839 if (startJobExecutor) {
840 this->runRuntimeJobExecutor();
841 }
842
843 while (!m_main_exit_flag.test(std::memory_order_acquire)) {
844 m_main_exit_flag.wait(false, std::memory_order_acquire);
845 }
846}

913template <template <class> class QueueType>
914void Dmn_Runtime_Manager<QueueType>::runPriorToCreateInstance() {
915 // Mask signals before creating any threads so they are blocked
916 // in all subsequently created threads (avoids races).
917 sigset_t old_mask{};
918 int err{};
919
920 sigemptyset(&Dmn_Runtime_Manager::s_mask);
921 sigaddset(&Dmn_Runtime_Manager::s_mask, SIGALRM);
922 sigaddset(&Dmn_Runtime_Manager::s_mask, SIGINT);
923 sigaddset(&Dmn_Runtime_Manager::s_mask, SIGTERM);
924 sigaddset(&Dmn_Runtime_Manager::s_mask, SIGQUIT);
925 sigaddset(&Dmn_Runtime_Manager::s_mask, SIGHUP);
926
927 err = pthread_sigmask(SIG_BLOCK, &Dmn_Runtime_Manager::s_mask, &old_mask);
928 if (0 != err) {
929 throw std::runtime_error("Error in pthread_sigmask: " +
930 std::string{strerror(err)});

946template <template <class> class QueueType>
947auto Dmn_Runtime_Manager<QueueType>::runRuntimeCoroutineScheduler() -> bool {
948 bool complete{true};
949
950 assert(isRunInAsyncThread());
951 assert(this->m_sched_task.size() > 0);
952 assert(this->m_sched_job.size() > 0);
953 assert(this->m_sched_task.size() == this->m_sched_job.size());
954
955 const Dmn_Runtime_Job &runningJob = this->m_sched_job.top();
956
957 auto &task = this->m_sched_task.top();
958
959 try {
960 if (task.isValid()) {
961 // runtime jobs must complete before returning to scheduler; suspension is
962 // only used for intra-job yielding and will be resumed immediately.
963 do {
964 task.m_handle.resume();
965
966 if (task.m_handle.done()) {
967 auto &ep = task.m_handle.promise().m_except;
968 if (ep && runningJob.m_onErrorFnc) {
969 std::rethrow_exception(ep);
970 }
971
972 break;
973 } else {
974 assert(!this->m_sched_job.empty());
975 assert(!this->m_sched_task.empty());
976
977 complete = false;
978 break;
979 }
980 } while (true);
981 }
982 } catch (...) {
983 try {
984 if (runningJob.m_onErrorFnc) {
985 std::exception_ptr ep = std::current_exception();
986
987 runningJob.m_onErrorFnc(ep);
988 }
989 } catch (...) {
990 // ignore it.
991 }
992 }
993
994 if (complete) {
995 this->m_sched_task.pop();
996 this->m_sched_job.pop();
997 }
998
999 return complete;
1000}
