454template <typename T, template <class> class QueueType>
455void Dmn_Pub<T, QueueType>::publishInternal(const T &item) {
456 /* Dmn_Async guarantees that publisher state is accessed by only one thread
457 * in the asynchronous context, so m_subscribers does not need a mutex.
458 * registerSubscriber() and unregisterSubscriber() run their mutations in
459 * that context and wait for the corresponding task before returning.
460 */
461
462 /* Keep the published item in circular ring buffer for
463 * efficient access to playback to new subscribers whose misses the
464 * data.
465 */
466
467 m_buffer.push_back(item);
468
469 if (m_buffer.size() > m_capacity) {
470 // Erase a range of elements from the start to the 'excess' point
471 // This performs all shifts in a single pass.
472 m_buffer.erase(m_buffer.begin(),
473 m_buffer.begin() + (m_buffer.size() - m_capacity));
474 }
475
476 std::exception_ptr firstFailure{};
477 for (auto &sub : m_subscribers) {
478 bool accepted = true;
479
480 if (m_filter_fn) {
481 try {
482 accepted = m_filter_fn(sub.get(), item);
483 } catch (...) {
484 auto failure = std::current_exception();
485 recordCallbackFailure(CallbackFailureKind::kFilter, *sub, failure);
486
487 if (!firstFailure) {
488 firstFailure = std::move(failure);
489 }
490
491 continue;
492 }
493 }
494
495 if (accepted) {
496 try {
497 sub->notify(item, this);
498 } catch (...) {
499 auto failure = std::current_exception();
500 recordCallbackFailure(CallbackFailureKind::kNotify, *sub, failure);
501
502 if (!firstFailure) {
503 firstFailure = std::move(failure);
504 }
505 }
506 }
507 }
508
509 if (firstFailure) {
510 std::rethrow_exception(firstFailure);
511 }
512} // method publishInternal()

514template <typename T, template <class> class QueueType>
515void Dmn_Pub<T, QueueType>::registerSubscriber(std::shared_ptr<Dmn_Sub> sub) {
516 auto waitHandler = this->addExecTaskWithWait([this, sub]() -> void {
517 auto it = std::find(sub->m_pubs.begin(), sub->m_pubs.end(), this);
518 if (it != sub->m_pubs.end()) {
519 return;
520 }
521
522 sub->m_pubs.push_back(this);
523 m_subscribers.push_back(sub);
524
525 // resend the data items that the registered subscriber
526 // miss.
527 size_t numberOfItemsToBeSkipped = 0;
528 if (sub->m_replayQuantity == 0) {
529 numberOfItemsToBeSkipped = m_buffer.size();
530 } else if (sub->m_replayQuantity > 0 &&
531 m_buffer.size() > static_cast<size_t>(sub->m_replayQuantity)) {
532 numberOfItemsToBeSkipped =
533 m_buffer.size() - static_cast<size_t>(sub->m_replayQuantity);
534 }
535
536 auto startIt = std::next(m_buffer.begin(), numberOfItemsToBeSkipped);
537 for (auto it = startIt; it != m_buffer.end(); it++) {
538 try {
539 sub->notify(*it, this);
540 } catch (...) {
541 recordCallbackFailure(CallbackFailureKind::kNotify, *sub,
542 std::current_exception());
543 }
544 }
545 });
546
547 waitHandler->wait();
548}
