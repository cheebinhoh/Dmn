426template <typename T, template <class> class QueueType>
427void Dmn_Pub<T, QueueType>::publishInternal(const T &item) {
428 /* Dmn_Async guarantees that publisher state is accessed by only one thread
429 * in the asynchronous context, so m_subscribers does not need a mutex.
430 * registerSubscriber() and unregisterSubscriber() run their mutations in
431 * that context and wait for the corresponding task before returning.
432 */
433
434 /* Keep the published item in circular ring buffer for
435 * efficient access to playback to new subscribers whose misses the
436 * data.
437 */
438
439 m_buffer.push_back(item);
440
441 if (m_buffer.size() > m_capacity) {
442 // Erase a range of elements from the start to the 'excess' point
443 // This performs all shifts in a single pass.
444 m_buffer.erase(m_buffer.begin(),
445 m_buffer.begin() + (m_buffer.size() - m_capacity));
446 }
447
448 std::exception_ptr firstFailure{};
449 for (auto &sub : m_subscribers) {
450 bool accepted = true;
451
452 if (m_filter_fn) {
453 try {
454 accepted = m_filter_fn(sub.get(), item);
455 } catch (...) {
456 auto failure = std::current_exception();
457 recordCallbackFailure(CallbackFailureKind::kFilter, *sub, failure);
458
459 if (!firstFailure) {
460 firstFailure = std::move(failure);
461 }
462
463 continue;
464 }
465 }
466
467 if (accepted) {
468 try {
469 sub->notify(item, this);
470 } catch (...) {
471 auto failure = std::current_exception();
472 recordCallbackFailure(CallbackFailureKind::kNotify, *sub, failure);
473
474 if (!firstFailure) {
475 firstFailure = std::move(failure);
476 }
477 }
478 }
479 }
480
481 if (firstFailure) {
482 std::rethrow_exception(firstFailure);
483 }
484} // method publishInternal()

486template <typename T, template <class> class QueueType>
487void Dmn_Pub<T, QueueType>::registerSubscriber(std::shared_ptr<Dmn_Sub> sub) {
488 auto waitHandler = this->addExecTaskWithWait([this, sub]() -> void {
489 auto it = std::find(sub->m_pubs.begin(), sub->m_pubs.end(), this);
490 if (it != sub->m_pubs.end()) {
491 return;
492 }
493
494 sub->m_pubs.push_back(this);
495 m_subscribers.push_back(sub);
496
497 // resend the data items that the registered subscriber
498 // miss.
499 size_t numberOfItemsToBeSkipped = 0;
500 if (sub->m_replayQuantity == 0) {
501 numberOfItemsToBeSkipped = m_buffer.size();
502 } else if (sub->m_replayQuantity > 0 &&
503 m_buffer.size() > static_cast<size_t>(sub->m_replayQuantity)) {
504 numberOfItemsToBeSkipped =
505 m_buffer.size() - static_cast<size_t>(sub->m_replayQuantity);
506 }
507
508 auto startIt = std::next(m_buffer.begin(), numberOfItemsToBeSkipped);
509 for (auto it = startIt; it != m_buffer.end(); it++) {
510 try {
511 sub->notify(*it, this);
512 } catch (...) {
513 recordCallbackFailure(CallbackFailureKind::kNotify, *sub,
514 std::current_exception());
515 }
516 }
517 });
518
519 waitHandler->wait();
520}
